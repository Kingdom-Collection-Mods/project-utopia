ut_new_dawn_fascist_bar = {
	name = "ut_new_dawn_fascist_bar"
	desc = "ut_new_dawn_fascist_bar_desc"

	#for color
	progress_texture = "gfx/interface/progressbar/progressbar_green.dds"
	background_texture = "gfx/interface/progressbar/progressbar_empty.dds"
	glow_texture = "gfx/interface/progressbar/progressbar_green_glow.dds"
	glow_line_texture = "gfx/interface/progressbar/progressbar_green_glow_line.dds"

	monthly_progress = {
		if = {
			limit = {
				NOT = {
					has_variable = ut_new_dawn_bar_freeze
				}
				has_variable = ut_nd_fascist_progress_from_characters
				has_variable = ut_nd_fascist_progress_from_igs
				has_variable = ut_nd_fascist_progress_from_legitimacy
				has_variable = ut_nd_fascist_progress_from_core_laws
				has_variable = ut_nd_fascist_progress_from_focus
				has_variable = ut_nd_fascist_progress_from_events
				has_variable = ut_nd_common_progress_from_events
			}
			add = {
				desc = "ut_nd_fascist_progress_from_characters"
				value = root.var:ut_nd_fascist_progress_from_characters
			}
			add = {
				desc = "ut_nd_fascist_progress_from_igs"
				value = root.var:ut_nd_fascist_progress_from_igs
			}
			add = {
				desc = "ut_nd_fascist_progress_from_legitimacy"
				value = root.var:ut_nd_fascist_progress_from_legitimacy
			}
			add = {
				desc = "ut_nd_fascist_progress_from_core_laws"
				value = root.var:ut_nd_fascist_progress_from_core_laws
			}
			add = {
				desc = "ut_nd_fascist_progress_from_focus"
				value = root.var:ut_nd_fascist_progress_from_focus
			}
			add = {
				desc = "ut_nd_fascist_progress_from_events"
				value = root.var:ut_nd_fascist_progress_from_events
				add = {
					value = root.var:ut_nd_common_progress_from_events
				}
			}
			if = {
				limit = {
					is_ai = yes
				}
				multiply = 2
			}
			else = {
				if ={
					limit = {
						OR = {
							has_game_rule = ut_new_dawn_start_era_4
							has_game_rule = ut_new_dawn_start_era_5
						}
						pop_type_percent_country = {
							target = pop_type:service_workers
							value < 0.01
						}
					}
					#no malus if starting era 4 or 5
				}
				else_if ={
					limit = {
						pop_type_percent_country = {
							target = pop_type:service_workers
							value < 0.01
						}
					}
					multiply = {
						desc = "ut_nd_multiplier_from_service_workers_0"
						value = 0.5
					}
				}
				else_if = {
					limit = {
						pop_type_percent_country = {
							target = pop_type:service_workers
							value < 0.02
						}
					}
					multiply = {
						desc = "ut_nd_multiplier_from_service_workers_1"
						value = 1.5
					}
				}
				else_if = {
					limit = {
						pop_type_percent_country = {
							target = pop_type:service_workers
							value < 0.03
						}
					}
					multiply = {
						desc = "ut_nd_multiplier_from_service_workers_2"
						value = 2
					}
				}
				else_if = {
					limit = {
						pop_type_percent_country = {
							target = pop_type:service_workers
							value < 0.04
						}
					}
					multiply = {
						desc = "ut_nd_multiplier_from_service_workers_3"
						value = 2.5
					}				
				}
				else = {
					multiply = {
						desc = "ut_nd_multiplier_from_service_workers_4"
						value = 3
					}
				}
			}
		}
		else_if = {
			limit = {
				je:je_ut_new_dawn = {
					OR = {
						"scripted_bar_progress(ut_new_dawn_fascist_bar)" >= 200
						AND = {
							"scripted_bar_progress(ut_new_dawn_fascist_bar)" >= 150
							AND = {
								"scripted_bar_progress(ut_new_dawn_communist_bar)" < 200
								"scripted_bar_progress(ut_new_dawn_democrat_bar)" < 200
								"scripted_bar_progress(ut_new_dawn_autocrat_bar)" < 200
							}
						}
						AND = {
							"scripted_bar_progress(ut_new_dawn_fascist_bar)" >= 100
							AND = {
								"scripted_bar_progress(ut_new_dawn_communist_bar)" < 150
								"scripted_bar_progress(ut_new_dawn_democrat_bar)" < 150
								"scripted_bar_progress(ut_new_dawn_autocrat_bar)" < 150
							}
						}
						AND = {
							"scripted_bar_progress(ut_new_dawn_fascist_bar)" >= 50
							AND = {
								"scripted_bar_progress(ut_new_dawn_communist_bar)" < 100
								"scripted_bar_progress(ut_new_dawn_democrat_bar)" < 100
								"scripted_bar_progress(ut_new_dawn_autocrat_bar)" < 100
							}
						}
						AND = {
							"scripted_bar_progress(ut_new_dawn_fascist_bar)" > 0
							AND = {
								"scripted_bar_progress(ut_new_dawn_communist_bar)" < 50
								"scripted_bar_progress(ut_new_dawn_democrat_bar)" < 50
								"scripted_bar_progress(ut_new_dawn_autocrat_bar)" < 50
							}
						}
					}
				}
			}
			add = {
				desc = "ut_nd_decaying_progress"
				value = -1
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 240 # new_dawn_core_upper_bound

}
