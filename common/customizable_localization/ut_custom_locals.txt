ut_new_dawn_progress_trend = {
	type = country
	random_valid = no

	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress >= 100
			var:je_ut_new_dawn_core_progress_diff >= 0
		}

		localization_key = ut_progress_trend_max
	}
	

	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress < 10
			var:je_ut_new_dawn_core_progress_diff < 0
		}

		localization_key = ut_progress_trend_critical
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			#probably some action happened like activating a milestone
			var:je_ut_new_dawn_core_progress_diff < -4
		}
		localization_key = ut_progress_trend_down_special
	}

	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < -2
		}

		localization_key = ut_progress_trend_down4
	}

	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < -1
		}

		localization_key = ut_progress_trend_down3
	}

	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < -0.7
		}

		localization_key = ut_progress_trend_down2
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < -0.4
		}

		localization_key = ut_progress_trend_down1
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < 0.1
			var:je_ut_new_dawn_core_progress_diff >  -0.1
		}

		localization_key = ut_progress_trend_neutral
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < 0.4
		}

		localization_key = ut_progress_trend_up1
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < 0.6
		}

		localization_key = ut_progress_trend_up2
	}
	
	text = {
		trigger = {
			has_variable = je_ut_new_dawn_core_progress_diff
			var:je_ut_new_dawn_core_progress_diff < 1
		}

		localization_key = ut_progress_trend_up3
	}

	text = {
		localization_key = ut_progress_trend_up4
	}

}
